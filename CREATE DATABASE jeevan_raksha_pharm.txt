CREATE TABLE Orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    order_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    total_amount DECIMAL(10,2) NOT NULL,
    payment_mode VARCHAR(20) NOT NULL,
    FOREIGN KEY (customer_id)
    REFERENCES Customers(customer_id),
    CHECK (payment_mode IN ('UPI', 'Cash', 'Card'))
);

# q1
SELECT customer_name, city
FROM Customers;

#q2
SELECT medicine_name
FROM Medicines
WHERE medicine_type IN ('Syrup', 'Injection');

#q3
SELECT *
FROM Orders
WHERE total_amount > 500;

#q4
SELECT phone
FROM Customers
WHERE city = 'Mumbai';

#q5
SELECT COUNT(*) AS medicine_count
FROM Medicines m
JOIN Suppliers s
    ON m.supplier_id = s.supplier_id
WHERE s.supplier_name = 'Apollo Distributors';

#q6
SELECT
    c.customer_name,
    o.order_date,
    o.total_amount
FROM Orders o
JOIN Customers c
    ON o.customer_id = c.customer_id;

#q7
SELECT
    payment_mode,
    SUM(total_amount) AS total_revenue
FROM Orders
GROUP BY payment_mode;

#q8
SELECT
    m.medicine_name,
    SUM(oi.quantity) AS total_quantity_sold
FROM Order_Items oi
JOIN Medicines m
    ON oi.medicine_id = m.medicine_id
GROUP BY m.medicine_id, m.medicine_name
ORDER BY total_quantity_sold DESC
LIMIT 1; 
#q9
SELECT
    c.customer_name,
    SUM(o.total_amount) AS total_spent
FROM Customers c
JOIN Orders o
    ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name
HAVING SUM(o.total_amount) > 1000;

#q10
SELECT
    m.medicine_name,
    m.stock_quantity,
    s.supplier_name
FROM Medicines m
JOIN Suppliers s
    ON m.supplier_id = s.supplier_id
WHERE m.stock_quantity < 50;
